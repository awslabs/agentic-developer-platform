"""Names/CPU, allowlisted AWS operation, TCP state and artifact metadata only."""
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import socket
import stat
import sys

args = json.loads(sys.stdin.readline())
assert set(args) == {"state_mount", "scope"}
scope = args["scope"]
assert scope["org_id"] == "a79914b7-45b6-4520-adeb-e09351267194"
assert scope["workspace_id"] == "f74c2f7b-a6c3-5d36-b512-41b86eda875a"
assert scope["operation_id"] == "b96cedbf-9afa-5cdf-9ee7-56e4b1e2e095"
report = {"observed_at": datetime.now(timezone.utc).isoformat(), "scope": scope, "processes": [], "artifacts": []}
report["cpu_cgroup"] = {}
for name in ("cpu.stat", "cpu.max"):
    try:
        report["cpu_cgroup"][name] = (Path("/sys/fs/cgroup") / name).read_text()
    except (FileNotFoundError, PermissionError):
        report["cpu_cgroup"][name] = None
ticks = os.sysconf("SC_CLK_TCK")
uptime = float(Path("/proc/uptime").read_text().split()[0])
aws_operations = {
    "kms": {"describe-key", "get-key-policy", "get-key-rotation-status", "list-aliases", "list-grants", "create-grant"},
    "iam": {"get-role", "get-policy", "get-policy-version", "list-attached-role-policies", "list-role-policies", "get-role-policy", "simulate-principal-policy", "get-instance-profile", "list-instance-profiles-for-role"},
    "sts": {"get-caller-identity", "assume-role"},
    "ec2": {"describe-subnets", "describe-route-tables", "describe-vpcs", "describe-security-groups", "describe-instance-types", "describe-account-attributes", "describe-availability-zones", "describe-images", "describe-instances"},
    "eks": {"describe-cluster", "list-access-entries", "describe-addon", "list-addons", "describe-access-entry"},
    "s3api": {"head-bucket", "get-bucket-location", "head-object", "list-objects-v2", "get-bucket-versioning", "get-bucket-encryption", "list-object-versions"},
    "dynamodb": {"describe-table", "get-item", "query"},
}
tcp = {}
states = {"01":"ESTABLISHED", "02":"SYN_SENT", "03":"SYN_RECV", "04":"FIN_WAIT1", "05":"FIN_WAIT2", "06":"TIME_WAIT", "07":"CLOSE", "08":"CLOSE_WAIT", "09":"LAST_ACK", "0A":"LISTEN", "0B":"CLOSING"}
for filename, family in (("tcp", socket.AF_INET), ("tcp6", socket.AF_INET6)):
    try:
        lines = (Path("/proc/net") / filename).read_text().splitlines()[1:]
    except (FileNotFoundError, PermissionError):
        continue
    for line in lines:
        columns = line.split()
        address, port = columns[2].split(":")
        raw = bytes.fromhex(address)
        raw = raw[::-1] if family == socket.AF_INET else b"".join(raw[i:i+4][::-1] for i in range(0,16,4))
        tcp[columns[9]] = {"remote_ip":socket.inet_ntop(family, raw), "remote_port":int(port,16), "tcp_state":states.get(columns[3], columns[3])}
for directory in Path("/proc").iterdir():
    if not directory.name.isdigit() or int(directory.name) == os.getpid():
        continue
    try:
        name = (directory / "comm").read_text().strip()
        fields = (directory / "stat").read_text().rsplit(") ", 1)[1].split()
        item = {"pid": int(directory.name), "name": name, "state": fields[0], "ppid": int(fields[1]), "user_cpu_seconds": int(fields[11])/ticks, "system_cpu_seconds": int(fields[12])/ticks, "start_ticks": int(fields[19]), "elapsed_seconds":round(uptime-int(fields[19])/ticks,3)}
        if name == "aws" or name.startswith("python"):
            argv = [(value.decode("utf-8", "replace")) for value in (directory/"cmdline").read_bytes().split(b"\x00") if value]
            if name == "aws":
                selected = [{"service":arg,"operation":argv[index+1]} for index,arg in enumerate(argv[:-1]) if arg in aws_operations and argv[index+1] in aws_operations[arg]]
                item["aws_command"] = selected[0] if len(selected)==1 else {"service":"unrecognized-redacted","operation":"unrecognized-redacted"}
            else:
                item["credential_process"] = any(Path(arg).name == "credential_process.py" for arg in argv)
            del argv
        item["tcp_sockets"], item["non_tcp_socket_count"] = [], 0
        for descriptor in (directory/"fd").iterdir():
            try:
                target = os.readlink(descriptor)
            except (FileNotFoundError, PermissionError):
                continue
            if target.startswith("socket:[") and target.endswith("]"):
                inode = target[8:-1]
                if inode in tcp:
                    item["tcp_sockets"].append(tcp[inode])
                else:
                    item["non_tcp_socket_count"] += 1
        report["processes"].append(item)
    except (FileNotFoundError, PermissionError, ProcessLookupError):
        continue
mount = Path(args["state_mount"])
assert mount.is_absolute() and ".." not in mount.parts and not mount.is_symlink()
key = hashlib.sha256(json.dumps([scope["org_id"], scope["workspace_id"], scope["operation_id"]], sort_keys=True, separators=(",", ":")).encode()).hexdigest()
private = mount / ("superplane-worker-" + str(os.getuid()))
assert not private.is_symlink()
root = private / key
assert not root.is_symlink()
report["operation_directory"] = str(root)


def metadata(path):
    try:
        value = path.lstat()
    except FileNotFoundError:
        return
    report["artifacts"].append({"path": str(path.relative_to(root)), "type": "symlink-not-followed" if stat.S_ISLNK(value.st_mode) else "directory" if stat.S_ISDIR(value.st_mode) else "file", "size_bytes": value.st_size, "modified_at": datetime.fromtimestamp(value.st_mtime, timezone.utc).isoformat()})


metadata(root)
for relative in (
    "bootstrap-state.json", "registrar", "installer", "supervisor", "workspace.tfvars.json", "module", "module/.terraform.lock.hcl", "module/.terraform",
    "module/.terraform/providers", "review", "review/workspace.tfplan",
    "review/workspace-plan.json", "review/workspace-authorization.proposed.json",
    "review/workspace-inventory.json", "review/workspace-estimate.json", "review/workspace-backend.json",
):
    selected = root / relative
    if any(parent.is_symlink() for parent in selected.parents if parent != root.parent):
        continue
    metadata(selected)
print(json.dumps(report))
